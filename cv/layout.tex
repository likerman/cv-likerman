\renewcommand{\cvsection}[1]{%

  {\color{gray}\hrule height 0.55pt}%

}

  \par\Needspace{3.5\baselineskip}%
  \addvspace{2mm}%

}

  \par\Needspace{3.8\baselineskip}%
  \addvspace{0mm}%

  \renewcommand{\arraystretch}{0.82}%
